-- Create these measures in SalesTable.

Total Revenue = SUM(SalesTable[revenue])

Total Profit = SUM(SalesTable[profit])

Profit Margin = DIVIDE([Total Profit], [Total Revenue])

Total Orders = DISTINCTCOUNT(SalesTable[order_id])

Average Order Value = DIVIDE([Total Revenue], [Total Orders])

Active Customers = DISTINCTCOUNT(SalesTable[customer_id])

Units Sold = SUM(SalesTable[quantity])

Previous Month Revenue =
VAR LastVisibleDate = MAX('CalendarTable'[date])
VAR IsFullMonth =
    HASONEVALUE('CalendarTable'[year_month])
        && COUNTROWS(VALUES('CalendarTable'[date])) = DAY(EOMONTH(LastVisibleDate, 0))
RETURN
    IF(
        IsFullMonth,
        CALCULATE(
            [Total Revenue],
            FILTER(
                ALL('CalendarTable'),
                'CalendarTable'[date] >= EOMONTH(LastVisibleDate, -2) + 1
                    && 'CalendarTable'[date] < EOMONTH(LastVisibleDate, -1) + 1
            )
        )
    )

Revenue MoM % =
VAR PreviousRevenue = [Previous Month Revenue]
RETURN
    DIVIDE(COALESCE([Total Revenue], 0) - PreviousRevenue, PreviousRevenue)
